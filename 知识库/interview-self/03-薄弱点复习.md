## 📞 Call/Apply/Bind 深度解析

### 核心概念对比

```javascript
// 三者的基本对比
function greet(greeting, punctuation) {
  return `${greeting}, ${this.name}${punctuation}`;
}

const person = { name: '张三' };

// call - 立即执行，逐个传参
const result1 = greet.call(person, 'Hello', '!'); // "Hello, 张三!"

// apply - 立即执行，数组传参
const result2 = greet.apply(person, ['Hi', '.']); // "Hi, 张三."

// bind - 返回新函数，延迟执行
const boundGreet = greet.bind(person, 'Hey');
const result3 = boundGreet('~'); // "Hey, 张三~"
```

### 1. Call 方法深入理解

#### 手写 Call 实现

```javascript
// 完整的 call 实现
Function.prototype.myCall = function(context, ...args) {
  // 1. 处理 context 为 null/undefined 的情况
  if (context === null || context === undefined) {
    context = window; // 浏览器环境
    // context = global; // Node.js 环境
  }

  // 2. 确保 context 是对象类型
  if (typeof context !== 'object') {
    context = Object(context);
  }

  // 3. 类型检查
  if (typeof this !== 'function') {
    throw new TypeError(`${this} is not a function`);
  }

  // 4. 使用 Symbol 创建唯一属性，避免覆盖原有属性
  const fnKey = Symbol('fn');

  // 5. 将函数挂载到 context 上
  context[fnKey] = this;

  // 6. 执行函数并收集结果
  let result;
  try {
    result = context[fnKey](...args);
  } finally {
    // 7. 清理临时属性，无论执行成功与否
    delete context[fnKey];
  }

  // 8. 返回执行结果
  return result;
};

// 测试用例
function testCall(name, age) {
  console.log(`${name}, ${this.msg}, ${age}岁`);
}

const obj = { msg: '你好' };

testCall.myCall(obj, '李四', 25); // "李四, 你好, 25岁"

// 边界情况测试
testCall.myCall(null, '全局', 30); // 全局环境下的 this 指向
testCall.myCall(123, '数字', 40); // 数字对象作为 this
```

### 2. Apply 方法深入理解

#### 手写 Apply 实现

```javascript
// 完整的 apply 实现
Function.prototype.myApply = function(context, argsArray) {
  // 1. 处理 context
  if (context === null || context === undefined) {
    context = window;
  }

  if (typeof context !== 'object') {
    context = Object(context);
  }

  // 2. 类型检查
  if (typeof this !== 'function') {
    throw new TypeError(`${this} is not a function`);
  }

  // 3. 处理参数数组
  let args = [];
  if (argsArray !== null && argsArray !== undefined) {
    // 检查是否为类数组对象
    if (!Array.isArray(argsArray) && !isArguments(argsArray)) {
      throw new TypeError('第二个参数必须是数组或类数组对象');
    }
    args = Array.from(argsArray);
  }

  // 4. 创建唯一属性并挂载函数
  const fnKey = Symbol('fn');
  context[fnKey] = this;

  // 5. 执行函数
  let result;
  try {
    result = context[fnKey](...args);
  } finally {
    // 6. 清理
    delete context[fnKey];
  }

  return result;
};

// 检查是否为 arguments 对象
function isArguments(obj) {
  return Object.prototype.toString.call(obj) === '[object Arguments]';
}

// 测试
function testApply(a, b, c) {
  console.log(this.value, a, b, c);
}

const obj2 = { value: '测试' };
testApply.myApply(obj2, [1, 2, 3]); // "测试 1 2 3"

// 使用场景：求数组最大值
const numbers = [5, 8, 2, 9, 1];
const max = Math.max.apply(null, numbers);
console.log(max); // 9

// 使用场景：合并数组
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const merged = Array.prototype.push.apply(arr1, arr2);
console.log(arr1); // [1, 2, 3, 4, 5, 6]
```

### 3. Bind 方法深入理解

#### 手写 Bind 实现

```javascript
// 完整的 bind 实现
Function.prototype.myBind = function(context, ...args) {
  // 1. 类型检查
  if (typeof this !== 'function') {
    throw new TypeError(`${this} is not a function`);
  }

  // 2. 保存原函数
  const self = this;

  // 3. 创建中转构造函数
  const fNOP = function() {};

  // 4. 创建绑定的函数
  const fBound = function(...newArgs) {
    // 判断是否作为构造函数调用
    const isNewCall = this instanceof fNOP;

    // 决定 this 指向
    const thisArg = isNewCall ? this : context;

    // 合并参数
    const allArgs = [...args, ...newArgs];

    // 执行原函数
    return self.apply(thisArg, allArgs);
  };

  // 5. 维护原型链
  if (this.prototype) {
    fNOP.prototype = this.prototype;
    fBound.prototype = new fNOP();
  }

  // 6. 返回绑定函数
  return fBound;
};

// 测试基本功能
function greet(name, age) {
  this.age = age;
  console.log(`我是${this.type}${name}，${age}岁`);
}

const person = { type: '学生' };
const boundGreet = greet.myBind(person, '张三');
boundGreet(20); // "我是学生张三，20岁"

// 测试构造函数调用
function Person(name, age) {
  this.name = name;
  this.age = age;
}

const BoundPerson = Person.myBind({ name: '默认' });
const p1 = new BoundPerson('李四', 25);
console.log(p1.name, p1.age); // "李四 25" (new 操作符优先级更高)

// 测试原型链
Person.prototype.say = function() {
  console.log(`我是${this.name}`);
};

const p2 = new BoundPerson('王五', 30);
p2.say(); // "我是王五" (原型链正常)
```

```javascript
// 3. 函数柯里化
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    } else {
      return function(...nextArgs) {
        return curried.apply(this, [...args, ...nextArgs]);
      };
    }
  };
}

function multiply(a, b, c) {
  return a * b * c;
}

const curriedMultiply = curry(multiply);
console.log(curriedMultiply(2)(3)(4)); // 24
console.log(curriedMultiply(2, 3)(4)); // 24
console.log(curriedMultiply(2)(3, 4)); // 24
```

### 继承的实现方式

#### 1. 原型链继承

```javascript
// 基本原型链继承
function Parent() {
  this.name = 'parent';
  this.colors = ['red', 'blue'];
}

Parent.prototype.sayName = function() {
  console.log(this.name);
};

function Child() {
  this.type = 'child';

// 继承：将 Parent 的实例作为 Child 的原型
Child.prototype = new Parent();

const child1 = new Child();
const child2 = new Child();
```

#### 2. 构造函数继承

```javascript

function Parent(name) {
  this.name = name;
  this.colors = ['red', 'blue'];

function Child(name) {
  // 继承属性：调用 Parent 构造函数
```

#### 3. 组合继承

```javascript
// 组合继承（原型链 + 构造函数）
function Parent(name) {
  this.name = name;
  this.colors = ['red', 'blue'];
}

Parent.prototype.sayName = function() {

function Child(name, age) {
```

#### 4. 寄生组合继承（最优方案）

```javascript
// 寄生组合继承
function inheritPrototype(child, parent) {
  // 创建父类原型的副本
  const prototype = Object.create(parent.prototype);

  // 修复 constructor
  prototype.constructor = child;

  // 将副本赋值给子类原型
  child.prototype = prototype;
}

function Parent(name) {
  this.name = name;
  this.colors = ['red', 'blue'];
}

Parent.prototype.sayName = function() {
  console.log(this.name);
};

function Child(name, age) {
  Parent.call(this, name);
  this.age = age;
}

// 使用寄生组合继承
inheritPrototype(Child, Parent);

Child.prototype.sayAge = function() {
  console.log(this.age);
};

const child = new Child('Jerry', 12);
console.log(child instanceof Child); // true
console.log(child instanceof Parent); // true
```

```javascript
// ES6 Class 继承
```

## 🔨 手写核心 API

### 1. 手写 Promise

```javascript
// 简化版 Promise 实现
class MyPromise {
  constructor(executor) {
    this.state = 'pending';
    this.value = undefined;
    this.reason = undefined;
    this.onFulfilledCallbacks = [];
    this.onRejectedCallbacks = [];

    const resolve = (value) => {
      if (this.state === 'pending') {
        this.state = 'fulfilled';
        this.value = value;
        this.onFulfilledCallbacks.forEach(fn => fn());
      }
    };

    const reject = (reason) => {
      if (this.state === 'pending') {
        this.state = 'rejected';
        this.reason = reason;
        this.onRejectedCallbacks.forEach(fn => fn());
      }
    };

    try {
      executor(resolve, reject);
    } catch (error) {
      reject(error);
    }
  }

  then(onFulfilled, onRejected) {
    onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : v => v;
    onRejected = typeof onRejected === 'function' ? onRejected : r => { throw r; };

    return new MyPromise((resolve, reject) => {
      const handleFulfilled = () => {
        if (this.state === 'fulfilled') {
          setTimeout(() => {
            try {
              const x = onFulfilled(this.value);
              resolve(x);
            } catch (error) {
              reject(error);
            }
          });
        }
      };

      const handleRejected = () => {
        if (this.state === 'rejected') {
          setTimeout(() => {
            try {
              const x = onRejected(this.reason);
              resolve(x);
            } catch (error) {
              reject(error);
            }
          });
        }
      };

      if (this.state === 'pending') {
        this.onFulfilledCallbacks.push(handleFulfilled);
        this.onRejectedCallbacks.push(handleRejected);
      } else {
        handleFulfilled();
        handleRejected();
      }
    });
  }

  catch(onRejected) {
    return this.then(null, onRejected);
  }

  finally(onFinally) {
    return this.then(
      value => MyPromise.resolve(onFinally()).then(() => value),
      reason => MyPromise.resolve(onFinally()).then(() => { throw reason })
    );
  }

  static resolve(value) {
    return new MyPromise(resolve => resolve(value));
  }

  static reject(reason) {
    return new MyPromise((resolve, reject) => reject(reason));
  }
}

// 测试
const promise = new MyPromise((resolve, reject) => {
  setTimeout(() => resolve('success'), 1000);
});

promise
  .then(value => console.log(value)) // "success"
  .catch(error => console.error(error))
  .finally(() => console.log('completed'));
```

### 2. 手写防抖和节流

```javascript
// 防抖：延迟执行，如果在延迟期间再次调用则重新计时
function debounce(func, delay, immediate = false) {
  let timeoutId;

  return function debounced(...args) {
    const callNow = immediate && !timeoutId;

    clearTimeout(timeoutId);

    timeoutId = setTimeout(() => {
      timeoutId = null;
      if (!immediate) {
        func.apply(this, args);
      }
    }, delay);

    if (callNow) {
      func.apply(this, args);
    }
  };
}

// 节流：固定时间间隔执行
function throttle(func, delay) {
  let lastCallTime = 0;
  let timeoutId;

  return function throttled(...args) {
    const now = Date.now();

    if (now - lastCallTime >= delay) {
      lastCallTime = now;
      func.apply(this, args);
    } else {
      clearTimeout(timeoutId);
      timeoutId = setTimeout(() => {
        lastCallTime = Date.now();
        func.apply(this, args);
      }, delay - (now - lastCallTime));
    }
  };
}

// 增强版节流：支持 leading 和 trailing
function throttleAdvanced(func, delay, options = {}) {
  let timeoutId;
  let lastCallTime = 0;
  let lastInvokeTime = 0;

  const { leading = true, trailing = true } = options;

  function invokeFunc(time) {
    const args = lastArgs;
    lastArgs = undefined;
    lastInvokeTime = time;
    func.apply(this, args);
  }

  function shouldInvoke(time) {
    const timeSinceLastCall = time - lastCallTime;
    const timeSinceLastInvoke = time - lastInvokeTime;

    return (
      lastCallTime === 0 ||
      timeSinceLastCall >= delay ||
      timeSinceLastCall < 0 ||
      (trailing && timeSinceLastInvoke >= delay)
    );
  }

  function trailingEdge(time) {
    timeoutId = undefined;
    if (trailing && lastArgs) {
      return invokeFunc(time);
    }
    lastArgs = undefined;
    return;
  }

  function timerExpired() {
    const time = Date.now();
    if (shouldInvoke(time)) {
      return trailingEdge(time);
    }
    const remaining = delay - (time - lastCallTime);
    timeoutId = setTimeout(timerExpired, remaining);
  }

  function throttled(...args) {
    const time = Date.now();
    const isInvoking = shouldInvoke(time);

    lastArgs = args;
    lastCallTime = time;

    if (isInvoking) {
      if (timeoutId === undefined) {
        if (leading) {
          lastInvokeTime = time;
          func.apply(this, args);
        }
      }
    }
    if (timeoutId === undefined) {
      timeoutId = setTimeout(timerExpired, delay);
    }
  }

  throttled.cancel = function() {
    if (timeoutId !== undefined) {
      clearTimeout(timeoutId);
    }
    lastArgs = undefined;
    lastCallTime = 0;
    lastInvokeTime = 0;
    timeoutId = undefined;
  };

  return throttled;
}

// 测试
const debouncedSearch = debounce((query) => {
  console.log('搜索:', query);
}, 300);

const throttledScroll = throttle(() => {
  console.log('滚动处理');
}, 100);

const advancedThrottle = throttleAdvanced((data) => {
  console.log('高级节流:', data);
}, 200, { leading: true, trailing: true });
```

### 3. 手写深拷贝

```javascript
// 深拷贝实现
function deepClone(obj, hash = new WeakMap()) {
  // 处理 null 或基本类型
  if (obj === null || typeof obj !== 'object') {
    return obj;
  }

  // 处理日期对象
  if (obj instanceof Date) {
    return new Date(obj.getTime());
  }

  // 处理正则表达式
  if (obj instanceof RegExp) {
    return new RegExp(obj.source, obj.flags);
  }

  // 处理 Map
  if (obj instanceof Map) {
    const clone = new Map();
    hash.set(obj, clone);
    obj.forEach((value, key) => {
      clone.set(key, deepClone(value, hash));
    });
    return clone;
  }

  // 处理 Set
  if (obj instanceof Set) {
    const clone = new Set();
    hash.set(obj, clone);
    obj.forEach(value => {
      clone.add(deepClone(value, hash));
    });
    return clone;
  }

  // 处理数组
  if (Array.isArray(obj)) {
    const clone = [];
    hash.set(obj, clone);
    obj.forEach((value, index) => {
      clone[index] = deepClone(value, hash);
    });
    return clone;
  }

  // 处理循环引用
  if (hash.has(obj)) {
    return hash.get(obj);
  }

  // 处理普通对象
  const clone = Object.create(Object.getPrototypeOf(obj));
  hash.set(obj, clone);

  // 处理 Symbol 属性
  const allKeys = [...Object.getOwnPropertyNames(obj), ...Object.getOwnPropertySymbols(obj)];

  allKeys.forEach(key => {
    clone[key] = deepClone(obj[key], hash);
  });

  return clone;
}

// 测试
const original = {
  name: 'test',
  date: new Date(),
  regex: /test/g,
  map: new Map([['key', 'value']]),
  set: new Set([1, 2, 3]),
  arr: [1, { nested: true }],
  [Symbol('test')]: 'symbol value'
};

original.self = original; // 循环引用

const cloned = deepClone(original);
console.log(cloned !== original); // true
console.log(cloned.self === cloned); // true
```

## ⚠️ 常见错误与陷阱

### 1. this 指向错误

```javascript
// 错误示例1：回调函数中的 this
const user = {
  name: 'Alice',
  friends: ['Bob', 'Charlie'],

  // ❌ 错误：this 指向丢失
  printFriendsBad() {
    this.friends.forEach(function(friend) {
      console.log(`${this.name}'s friend: ${friend}`); // this.name 为 undefined
    });
  },

  // ✅ 正确：使用箭头函数
  printFriendsGood() {
    this.friends.forEach(friend => {
      console.log(`${this.name}'s friend: ${friend}`); // 正确
    });
  },

  // ✅ 正确：保存 this
  printFriendsAlternative() {
    const self = this;
    this.friends.forEach(function(friend) {
      console.log(`${self.name}'s friend: ${friend}`); // 正确
    });
  }
};
```

### 2. 原型链污染

```javascript
// 错误示例：原型链污染
function Person() {}

// ❌ 危险：在原型上添加可枚举属性
Person.prototype.isAdmin = false;

const person = new Person();

// 遍历对象时会出现问题
for (let key in person) {
  console.log(key); // "isAdmin" - 可能不是期望的
}

// ✅ 正确：使用 Object.defineProperty
Object.defineProperty(Person.prototype, 'isAdmin', {
  value: false,
  enumerable: false, // 不可枚举
  writable: true,
  configurable: true
});
```

### 3. bind 的多次调用

```javascript
// bind 的多次调用
function greet() {
  console.log(this.name);
}

const obj1 = { name: 'obj1' };
const obj2 = { name: 'obj2' };

// 第一次 bind
const bound1 = greet.bind(obj1);

// 第二次 bind（无效）
const bound2 = bound1.bind(obj2);

bound1(); // "obj1" - 正常
bound2(); // "obj1" - 第二次 bind 无效，仍然指向 obj1
```

### 4. call/apply 的类型错误

```javascript
// 错误示例：call/apply 的第一个参数不是对象
function showThis() {
  console.log(this);
}

// 原始类型会被自动包装
showThis.call('string'); // String 对象
showThis.call(123); // Number 对象
showThis.call(true); // Boolean 对象

// null 和 undefined 在非严格模式下指向 window
showThis.call(null); // window
showThis.call(undefined); // window

// 严格模式下
'use strict';
showThis.call(null); // null
showThis.call(undefined); // undefined
```
