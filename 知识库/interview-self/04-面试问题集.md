#### Q1: JavaScript 中有哪些数据类型？如何准确判断类型？

```javascript
// 基本数据类型
let num = 123;           // number
let str = "hello";       // string
let bool = true;         // boolean
let nullVar = null;      // null
let undefinedVar;        // undefined
let sym = Symbol();      // symbol

// 引用数据类型
let obj = {};            // object
let arr = [];            // object (array)

console.log(getType({}));        // "Object"
console.log(getType(null));      // "Null"
console.log(getType(undefined)); // "Undefined"

console.log(getType("hello"));   // "String"
```

#### Q2: 解释 == 和 === 的区别，以及类型转换规则

```javascript
// === 严格相等：值和类型都必须相同
console.log(1 === "1");   // false
console.log(true === 1);  // false
console.log(null === undefined); // false

// == 抽象相等：会发生类型转换
console.log(1 == "1");    // true  (字符串转数字)
console.log(true == 1);   // true  (布尔值转数字)
console.log(null == undefined); // true (特殊情况)

// 类型转换规则
// 1. 数字和字符串：字符串转数字
// 2. 布尔值和其他类型：布尔值转数字 (true→1, false→0)
// 3. 对象和原始类型：对象转原始值
// 4. null 和 undefined 相等但不严格相等
// 5. NaN 不等于任何值，包括自身

// 特殊情况
console.log(NaN == NaN);     // false
console.log(NaN === NaN);    // false
console.log(0 == -0);        // true
console.log(0 === -0);       // true
```

### 作用域与闭包

#### Q3: 什么是闭包？闭包的实际应用场景有哪些？

```javascript
// 闭包：函数可以访问其外部作用域的变量
function outer(x) {
  // 外部函数的变量
  return function inner(y) {
    // 内部函数可以访问外部变量 x
    return x + y;
  };
}

const add5 = outer(5);
console.log(add5(3)); // 8

// 实际应用场景

// 1. 模块化 - 避免全局污染
const counter = (function() {
  let count = 0;

  return {
    increment() {
      count++;
      return count;
    },
    decrement() {
      count--;
      return count;
    },
    getCount() {
      return count;
    }
  };
})();

console.log(counter.increment()); // 1
console.log(counter.getCount());  // 1

// 2. 函数柯里化
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    } else {
      return function(...nextArgs) {
        return curried.apply(this, [...args, ...nextArgs]);
      };
    }
  };
}

const add = (a, b, c) => a + b + c;
const curriedAdd = curry(add);
console.log(curriedAdd(1)(2)(3)); // 6

// 3. 防抖和节流
function debounce(func, delay) {
  let timeoutId;
  return function(...args) {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => {
      func.apply(this, args);
    }, delay);
  };
}

// 4. 事件处理中的状态保持
function attachButtonListeners() {
  const buttons = document.querySelectorAll('button');

  buttons.forEach((button, index) => {
    button.addEventListener('click', function() {
      // 闭包保存了 index 的值
      console.log(`Button ${index} clicked`);
    });
  });
}
```

### this 指向

#### Q5: this 的绑定规则有哪些？优先级如何？

```javascript
// this 的四种绑定规则，按优先级排序

// 1. new 绑定 - 最高优先级
function Person(name) {
  this.name = name;
}

const person = new Person("Alice");
console.log(person.name); // "Alice"

// 2. 显式绑定 - call/apply/bind
function greet() {
  console.log(`Hello, ${this.name}`);
}

const obj1 = { name: "Bob" };
const obj2 = { name: "Charlie" };

greet.call(obj1);    // "Hello, Bob"
greet.apply(obj2);   // "Hello, Charlie"

const boundGreet = greet.bind(obj1);
boundGreet();        // "Hello, Bob"

// 3. 隐式绑定 - 对象方法调用
const obj3 = {
  name: "David",
  greet() {
    console.log(`Hello, ${this.name}`);
  }
};

obj3.greet();       // "Hello, David"

// 隐式丢失
const greetFunc = obj3.greet;
greetFunc();        // "Hello, undefined" (非严格模式) 或 TypeError (严格模式)

// 4. 默认绑定 - 最低优先级
function showThis() {
  console.log(this);
}

showThis(); // window (非严格模式) 或 undefined (严格模式)

// 优先级测试
function testPriority() {
  console.log(this.name);
}

const obj4 = { name: "Priority Test" };

// 显式绑定 vs 隐式绑定
obj4.method = testPriority;
obj4.method.call({ name: "Explicit" }); // "Explicit" - 显式绑定优先

// new 绑定 vs 显式绑定
const BoundTest = testPriority.bind({ name: "Bound" });
const instance = new BoundTest(); // undefined - new 绑定优先
```

### 异步编程

#### Q6: Promise 的状态和常用方法有哪些？如何处理异步错误？

```javascript
// Promise 的三种状态
const promise = new Promise((resolve, reject) => {
  // pending -> fulfilled
  // pending -> rejected
  // 状态一旦确定就不能改变
});

// 常用方法

// 1. Promise.resolve()
Promise.resolve(42);              // Promise { 42 }
Promise.resolve(new Promise(() => {})); // 返回原 Promise

// 2. Promise.reject()
Promise.reject(new Error("error"));

// 3. Promise.all() - 全部成功才成功
Promise.all([
  Promise.resolve(1),
  Promise.resolve(2),
  Promise.resolve(3)
]).then(results => console.log(results)); // [1, 2, 3]

// 4. Promise.race() - 第一个完成的结果
Promise.race([
  Promise.resolve(1),
  new Promise(resolve => setTimeout(() => resolve(2), 100))
]).then(result => console.log(result)); // 1

// 5. Promise.allSettled() - 等待全部完成
Promise.allSettled([
  Promise.resolve(1),
  Promise.reject(new Error("failed")),
  Promise.resolve(3)
]).then(results => {
  console.log(results);
  // [
  //   { status: 'fulfilled', value: 1 },
  //   { status: 'rejected', reason: Error },
  //   { status: 'fulfilled', value: 3 }
  // ]
});

// 错误处理
function fetchUser(id) {
  return fetch(`/api/users/${id}`)
    .then(response => {
      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }
      return response.json();
    })
    .catch(error => {
      // 处理网络错误和上面的错误
      console.error("Failed to fetch user:", error);
      throw error; // 重新抛出错误，让调用者处理
    });
}

// 使用 async/await 处理错误
async function getUser(id) {
  try {
    const user = await fetchUser(id);
    console.log("User:", user);
    return user;
  } catch (error) {
    console.error("Error in getUser:", error);
    return null; // 返回默认值
  }
}
```

### 类型系统

#### Q7: TypeScript 中的 interface 和 type 有什么区别？

```typescript
// 相同点：都可以定义对象类型
interface UserInterface {
  name: string;
  age: number;
}

type UserType = {
  name: string;
  age: number;
};

// 不同点：

// 1. 继承方式不同
interface Person {
  name: string;
}

interface Employee extends Person {
  employeeId: string;
}

// type 使用交叉类型继承
type Animal = {
  name: string;
};

type Dog = Animal & {
  breed: string;
};

// 2. 声明合并
interface User {
  name: string;
}

interface User {
  age: number; // 自动合并
}

// type 不能重复声明
// type User = {
//   email: string;
// }; // Error: Duplicate identifier 'User'

// 3. 联合类型和交叉类型
type Status = 'pending' | 'success' | 'error'; // 联合类型
type AdminUser = UserInterface & { isAdmin: true }; // 交叉类型

// interface 不能直接表示联合类型

// 4. 复杂类型表达式
type Keys = keyof UserInterface; // "name" | "age"
type PartialUser = Partial<UserInterface>; // 所有属性可选

// interface 无法这样使用

// 5. 元组类型
type Point = [number, number]; // 元组类型

// interface 不能定义元组

// 使用建议：
// - 优先使用 interface 定义对象类型和类的结构
// - 使用 type 定义联合类型、交叉类型、工具类型
// - 需要声明合并时使用 interface
```

#### Q8: 什么是泛型？泛型的约束和条件类型如何使用？

```typescript
// 基本泛型
function identity<T>(arg: T): T {
  return arg;
}

const numResult = identity<number>(42);
const strResult = identity("hello");

// 泛型约束
interface Lengthwise {
  length: number;
}

function logLength<T extends Lengthwise>(arg: T): void {
  console.log(`Length: ${arg.length}`);
}

logLength("hello");     // OK
logLength([1, 2, 3]);   // OK
// logLength(42);       // Error: number 没有 length 属性

// 多个泛型参数
function pair<T, U>(first: T, second: U): [T, U] {
  return [first, second];
}

const [str, num] = pair("hello", 42);

// 条件类型
type IsString<T> = T extends string ? true : false;

type Test1 = IsString<string>;  // true
type Test2 = IsString<number>;  // false

// 实用条件类型
type NonNullable<T> = T extends null | undefined ? never : T;

type Test3 = NonNullable<string | null>;  // string
type Test4 = NonNullable<number | undefined>; // number

// 映射类型
type Readonly<T> = {
  readonly [P in keyof T]: T[P];
};

type Partial<T> = {
  [P in keyof T]?: T[P];
};

// 使用示例
interface Todo {
  title: string;
  description: string;
  completed: boolean;
}

type PartialTodo = Partial<Todo>;
type ReadonlyTodo = Readonly<Todo>;

// 高级映射类型
type Stringify<T> = {
  [K in keyof T]: T[K] extends string ? T[K] : `${T[K]}`;
};

// 泛型工具类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;
type Parameters<T> = T extends (...args: infer P) => any ? P : never;

function greet(name: string): string {
  return `Hello, ${name}`;
}

type GreetReturn = ReturnType<typeof greet>;        // string
type GreetParams = Parameters<typeof greet>;       // [string]
```

### 高级类型

#### Q9: 解释 infer 关键字的作用和应用场景

```typescript
// infer 用于在条件类型中推断类型

// 1. 推断函数返回类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never;

function getUser() {
  return { id: 1, name: "Alice" };
}

type User = ReturnType<typeof getUser>; // { id: number; name: string; }

// 2. 推断数组元素类型
type ArrayElement<T> = T extends (infer U)[] ? U : never;

type StringArray = string[];
type Element = ArrayElement<StringArray>; // string

// 3. 推解 Promise 的值类型
type UnpackPromise<T> = T extends Promise<infer U> ? U : never;

type StringPromise = Promise<string>;
type Unpacked = UnpackPromise<StringPromise>; // string

// 4. 推解函数参数类型
type FirstParameter<T> = T extends (arg1: infer A, ...args: any[]) => any ? A : never;

function processData(data: string, options: any): void {}
type DataParam = FirstParameter<typeof processData>; // string

// 5. 推解对象值的类型
type ValueOf<T> = T extends { [key: string]: infer V } ? V : never;

type Config = {
  apiUrl: string;
  timeout: number;
  retry: boolean;
};

type ConfigValue = ValueOf<Config>; // string | number | boolean

// 6. 构造函数推断
type InstanceType<T> = T extends new (...args: any[]) => infer R ? R : never;

class User {
  constructor(public name: string) {}
}

type UserInstance = InstanceType<typeof User>; // User

// 7. 元组推断
type Head<T> = T extends [infer H, ...any[]] ? H : never;
type Tail<T> = T extends [any, ...infer T] ? T : never;

type Tuple = [string, number, boolean];
type First = Head<Tuple>;  // string
type Rest = Tail<Tuple>;   // [number, boolean]

// 实际应用：深度只读
type DeepReadonly<T> = {
  readonly [P in keyof T]: T[P] extends object ? DeepReadonly<T[P]> : T[P];
};

interface Nested {
  user: {
    name: string;
    settings: {
      theme: string;
    };
  };
}

type DeepReadonlyNested = DeepReadonly<Nested>;
// 所有嵌套属性都变成只读
```

#### Q10: useEffect 的执行时机和清理函数如何工作？

```jsx
import React, { useState, useEffect } from 'react';

function useEffectDemo() {
  const [count, setCount] = useState(0);
  const [name, setName] = useState('');

  // 1. 没有依赖数组 - 每次渲染后都执行
  useEffect(() => {
    console.log('每次渲染后执行');
  });

  // 2. 空依赖数组 - 只在组件挂载时执行一次
  useEffect(() => {
    console.log('组件挂载时执行');

    // 清理函数在组件卸载时执行
    return () => {
      console.log('组件卸载时清理');
    };
  }, []);

  // 3. 有依赖数组 - 依赖变化时执行
  useEffect(() => {
    console.log('count 变化:', count);

    // 清理函数在下一个 effect 执行前执行
    return () => {
      console.log('清理上一个 effect');
    };
  }, [count]);

  // 4. 多个依赖
  useEffect(() => {
    console.log('count 或 name 变化');
  }, [count, name]);

  // 实际应用：订阅外部数据源
  useEffect(() => {
    const timer = setInterval(() => {
      setCount(prev => prev + 1);
    }, 1000);

    // 清理定时器
    return () => {
      clearInterval(timer);
    };
  }, []);

  // 实际应用：事件监听
  useEffect(() => {
    const handleScroll = () => {
      console.log('滚动位置:', window.scrollY);
    };

    window.addEventListener('scroll', handleScroll);

    return () => {
      window.removeEventListener('scroll', handleScroll);
    };
  }, []);

  // 实际应用：API 请求
  useEffect(() => {
    if (name) {
      fetch(`/api/users?name=${name}`)
        .then(response => response.json())
        .then(data => console.log('搜索结果:', data))
        .catch(error => console.error('搜索失败:', error));
    }
  }, [name]);

  return (
    <div>
      <p>Count: {count}</p>
      <button onClick={() => setCount(count + 1)}>增加</button>

      <input
        value={name}
        onChange={(e) => setName(e.target.value)}
        placeholder="搜索用户"
      />
    </div>
  );
}
```

#### Q11: React.memo、useMemo、useCallback 的使用场景和区别？

```jsx
import React, { useState, memo, useMemo, useCallback } from 'react';

// React.memo - 组件级别的优化
const ExpensiveComponent = memo(({ data, onUpdate }) => {
  console.log('ExpensiveComponent 渲染');

  // 模拟昂贵的计算
  const processedData = useMemo(() => {
    console.log('处理数据...');
    return data.map(item => ({
      ...item,
      processed: true,
      value: item.value * 2
    }));
  }, [data]);

  return (
    <div>
      {processedData.map(item => (
        <div key={item.id}>
          {item.name}: {item.value}
        </div>
      ))}
      <button onClick={() => onUpdate(data[0])}>
        更新第一项
      </button>
    </div>
  );
});

// 父组件
function ParentComponent() {
  const [count, setCount] = useState(0);
  const [data, setData] = useState([
    { id: 1, name: 'Item 1', value: 10 },
    { id: 2, name: 'Item 2', value: 20 }
  ]);

  // ❌ 错误：每次渲染都创建新函数
  const handleUpdateBad = (item) => {
    setData(prevData =>
      prevData.map(d => d.id === item.id ? { ...d, value: d.value + 1 } : d)
    );
  };

  // ✅ 正确：使用 useCallback 缓存函数
  const handleUpdate = useCallback((item) => {
    setData(prevData =>
      prevData.map(d => d.id === item.id ? { ...d, value: d.value + 1 } : d)
    );
  }, []); // 空依赖数组，因为使用了函数式更新

  // ✅ 复杂计算使用 useMemo
  const statistics = useMemo(() => {
    console.log('计算统计信息...');
    const total = data.reduce((sum, item) => sum + item.value, 0);
    const average = total / data.length;
    const max = Math.max(...data.map(item => item.value));
    return { total, average, max };
  }, [data]);

  // ✅ 事件处理函数使用 useCallback
  const handleAddItem = useCallback(() => {
    const newItem = {
      id: data.length + 1,
      name: `Item ${data.length + 1}`,
      value: Math.floor(Math.random() * 100)
    };
    setData(prevData => [...prevData, newItem]);
  }, [data.length]);

  return (
    <div>
      <h3>统计信息</h3>
      <p>总计: {statistics.total}</p>
      <p>平均: {statistics.average.toFixed(2)}</p>
      <p>最大值: {statistics.max}</p>

      <h3>数据列表</h3>
      <ExpensiveComponent data={data} onUpdate={handleUpdate} />

      <button onClick={handleAddItem}>添加项目</button>
      <button onClick={() => setCount(count + 1)}>
        重新渲染 (Count: {count})
      </button>
    </div>
  );
}

// 自定义比较函数
const UserCard = memo(({ user, onSelect }) => {
```

### 状态管理

#### Q12: React 中的状态管理模式有哪些？各自的优缺点是什么？

```jsx
// 1. useState - 基本状态管理
function LocalStateComponent() {
  const [user, setUser] = useState({ name: '', age: 0 });
  const [loading, setLoading] = useState(false);

  const updateUser = (newUser) => {
    setUser(newUser);
  };

  // 优点：简单直接
  // 缺点：状态分散，难以共享
}

// 2. useReducer - 复杂状态管理
function reducer(state, action) {
  switch (action.type) {
    case 'SET_LOADING':
      return { ...state, loading: action.payload };
    case 'SET_USER':
      return { ...state, user: action.payload, loading: false };
    case 'SET_ERROR':
      return { ...state, error: action.payload, loading: false };
    default:
      return state;
  }
}

function ReducerComponent() {
  const [state, dispatch] = useReducer(reducer, {
    user: null,
    loading: false,
    error: null
  });

  const fetchUser = async (id) => {
    dispatch({ type: 'SET_LOADING', payload: true });

    try {
      const response = await fetch(`/api/users/${id}`);
      const user = await response.json();
      dispatch({ type: 'SET_USER', payload: user });
    } catch (error) {
      dispatch({ type: 'SET_ERROR', payload: error.message });
    }
  };

  // 优点：逻辑集中，可预测性强
  // 缺点：代码量较多，学习成本高
}

// 3. Context API - 全局状态管理
const UserContext = React.createContext();
```

```jsx
// 虚拟滚动实现
function VirtualList({
  items = [],
  itemHeight = 50,
  containerHeight = 400,
  renderItem,
  overscan = 5
}) {
  const [scrollTop, setScrollTop] = useState(0);
  const containerRef = useRef(null);

  // 计算可见范围
  const visibleRange = useMemo(() => {
    const startIndex = Math.max(0, Math.floor(scrollTop / itemHeight) - overscan);
    const endIndex = Math.min(
      items.length - 1,
      Math.ceil((scrollTop + containerHeight) / itemHeight) + overscan
    );

    return { startIndex, endIndex };
  }, [scrollTop, itemHeight, containerHeight, items.length, overscan]);

  // 可见项目
  const visibleItems = useMemo(() => {
    return items.slice(visibleRange.startIndex, visibleRange.endIndex + 1);
  }, [items, visibleRange]);

  const handleScroll = (e) => {
    setScrollTop(e.currentTarget.scrollTop);
  };

  return (
    <div
      ref={containerRef}
      style={{
        height: containerHeight,
        overflow: 'auto',
        border: '1px solid #ccc'
      }}
      onScroll={handleScroll}
    >
      <div style={{ height: items.length * itemHeight, position: 'relative' }}>
        {visibleItems.map((item, index) => {
          const actualIndex = visibleRange.startIndex + index;
          return (
            <div
              key={item.id || actualIndex}
              style={{
                position: 'absolute',
                top: actualIndex * itemHeight,
                height: itemHeight,
                width: '100%',
                boxSizing: 'border-box'
              }}
            >
              {renderItem(item, actualIndex)}
            </div>
          );
        })}
      </div>
    </div>
  );
}

function DynamicVirtualList({ items, containerHeight, renderItem }) {
  const [itemHeights, setItemHeights] = useState({});
  const [scrollTop, setScrollTop] = useState(0);
  const containerRef = useRef(null);

  // 获取项目高度
  const getItemHeight = useCallback((index) => {
    return itemHeights[index] || 50; // 默认高度
  }, [itemHeights]);

  // 计算项目偏移量
  const getItemOffset = useCallback((index) => {
    let offset = 0;
    for (let i = 0; i < index; i++) {
      offset += getItemHeight(i);
    }
    return offset;
  }, [getItemHeight]);

  // 计算总高度
  const totalHeight = useMemo(() => {
    return items.reduce((total, _, index) => total + getItemHeight(index), 0);
  }, [items, getItemHeight]);

  // 更新项目高度
  const setItemHeight = useCallback((index, height) => {
    setItemHeights(prev => ({
      ...prev,
      [index]: height
    }));
  }, []);

  return (
    <div
      ref={containerRef}
      style={{
        height: containerHeight,
        overflow: 'auto'
      }}
      onScroll={(e) => setScrollTop(e.currentTarget.scrollTop)}
    >
      <div style={{ height: totalHeight, position: 'relative' }}>
        {items.map((item, index) => {
          const offset = getItemOffset(index);
          const height = getItemHeight(index);

          // 简单的可见性检查
          if (offset + height < scrollTop || offset > scrollTop + containerHeight) {
            return null;
          }

          return (
            <div
              key={item.id}
              style={{
                position: 'absolute',
                top: offset,
                width: '100%'
              }}
            >
              <div ref={(el) => {
                if (el) {
                  const newHeight = el.offsetHeight;
                  if (newHeight !== getItemHeight(index)) {
                    setItemHeight(index, newHeight);
                  }
                }
              }}>
                {renderItem(item, index)}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
```

```javascript
// 冒泡排序
function bubbleSort(arr) {
  const result = [...arr];
  const n = result.length;

  for (let i = 0; i < n - 1; i++) {
    for (let j = 0; j < n - i - 1; j++) {
      if (result[j] > result[j + 1]) {
        [result[j], result[j + 1]] = [result[j + 1], result[j]];
      }
    }
  }

  return result;
}

// 快速排序
function quickSort(arr) {
  if (arr.length <= 1) return arr;

  const pivot = arr[Math.floor(arr.length / 2)];
  const left = arr.filter(x => x < pivot);
  const middle = arr.filter(x => x === pivot);
  const right = arr.filter(x => x > pivot);

  return [...quickSort(left), ...middle, ...quickSort(right)];
}

// 归并排序
function mergeSort(arr) {
  if (arr.length <= 1) return arr;

  const mid = Math.floor(arr.length / 2);
  const left = mergeSort(arr.slice(0, mid));
  const right = mergeSort(arr.slice(mid));

  return merge(left, right);
}

function merge(left, right) {
  const result = [];
  let i = 0, j = 0;

  while (i < left.length && j < right.length) {
    if (left[i] <= right[j]) {
      result.push(left[i++]);
    } else {
      result.push(right[j++]);
    }
  }

  return [...result, ...left.slice(i), ...right.slice(j)];
}

// 对象数组排序
function sortObjects(arr, key, order = 'asc') {
  return [...arr].sort((a, b) => {
    const aVal = a[key];
    const bVal = b[key];

    if (order === 'desc') {
      return bVal > aVal ? 1 : bVal < aVal ? -1 : 0;
    }

    return aVal > bVal ? 1 : aVal < bVal ? -1 : 0;
  });
}

// 多字段排序
function multiSort(arr, sortFields) {
  return [...arr].sort((a, b) => {
    for (const { key, order = 'asc' } of sortFields) {
      const aVal = a[key];
      const bVal = b[key];

      if (aVal !== bVal) {
        const comparison = aVal > bVal ? 1 : -1;
        return order === 'desc' ? -comparison : comparison;
      }
    }
    return 0;
  });
}
```

```javascript
function advancedTemplate(template, data) {
  return template.replace(/\{\{([^}]+)\}\}/g, (match, expression) => {
    try {
      // 处理嵌套属性
      const keys = expression.trim().split('.');
      let value = data;

      for (const key of keys) {
        value = value?.[key];
      }

      return value !== undefined ? value : match;
    } catch (error) {
      return match;
    }
  });
}

function conditionalTemplate(template, data) {
  // 处理 if 语句 {{#if condition}}...{{/if}}
  return template.replace(/\{\{#if\s+(\w+)\}\}([\s\S]*?)\{\{\/if\}\}/g, (match, condition, content) => {
    return data[condition] ? content : '';
  });
}

// 支持循环语句
function loopTemplate(template, data) {
  // 处理 each 语句 {{#each array}}...{{/each}}
  return template.replace(/\{\{#each\s+(\w+)\}\}([\s\S]*?)\{\{\/each\}\}/g, (match, arrayName, content) => {
    const array = data[arrayName] || [];
    return array.map((item, index) => {
      return content
        .replace(/\{\{this\}\}/g, item)
        .replace(/\{\{@index\}\}/g, index)
        .replace(/\{\{(\w+)\}\}/g, (subMatch, key) => item[key] || '');
    }).join('');
  });
}

  result = conditionalTemplate(result, data);

  // 再处理循环语句
  result = loopTemplate(result, data);

  // 最后处理简单替换
  result = advancedTemplate(result, data);

  return result;
}
```

```jsx
            if (imageData) {

      this.init();

      onLoad?.();
      this.images.delete(element);

      onError?.();
      this.images.delete(element);

      this.observer.unobserve(element);
    }
    this.images.delete(element);
  }

  destroy() {
    if (this.observer) {
      this.observer.disconnect();
      this.observer = null;
    }
    this.images.clear();
  }
}
```

```jsx
class Command {
  constructor(execute, undo, description) {
    this.execute = execute;
    this.undo = undo;
    this.description = description;

// 历史记录管理器
class HistoryManager {
  constructor(maxSize = 50) {
    this.history = [];
    this.currentIndex = -1;
    this.maxSize = maxSize;
  }

  execute(command) {
    // 清除当前位置之后的历史记录
    this.history = this.history.slice(0, this.currentIndex + 1);

    // 执行命令
    command.execute();

    // 添加到历史记录
    this.history.push(command);
    this.currentIndex++;

    // 限制历史记录大小
    if (this.history.length > this.maxSize) {
      this.history.shift();
      this.currentIndex--;
    }
  }

  undo() {
    if (this.canUndo()) {
      const command = this.history[this.currentIndex];
      command.undo();
      this.currentIndex--;
      return true;
    }
    return false;
  }

  redo() {
    if (this.canRedo()) {
      this.currentIndex++;
      const command = this.history[this.currentIndex];
      command.execute();
      return true;
    }
    return false;
  }

  canUndo() {
    return this.currentIndex >= 0;
  }

  canRedo() {
    return this.currentIndex < this.history.length - 1;
  }

  getHistory() {
    return this.history.map((cmd, index) => ({
      description: cmd.description,
      active: index <= this.currentIndex
    }));
  }

  clear() {
    this.history = [];
    this.currentIndex = -1;
  }
}

    );

    executeCommand(command);
  }, [getSelection, setSelectionRange, executeCommand]);

  // 删除文本命令
  const deleteText = useCallback((start, end) => {
    const actualStart = Math.min(start, end);
    const actualEnd = Math.max(start, end);
    const deletedText = content.substring(actualStart, actualEnd);

    const command = new Command(
      // execute
      () => {
        const textarea = textareaRef.current;
        const beforeText = textarea.value.substring(0, actualStart);
        const afterText = textarea.value.substring(actualEnd);
        const newText = beforeText + afterText;

        textarea.value = newText;
        setContent(newText);
        setSelectionRange(actualStart, actualStart);
      },
      // undo
      () => {
        const textarea = textareaRef.current;
        const beforeText = textarea.value.substring(0, actualStart);
        const afterText = textarea.value.substring(actualStart);
        const oldText = beforeText + deletedText + afterText;

        textarea.value = oldText;
        setContent(oldText);
        setSelectionRange(actualEnd, actualEnd);
      },
      `删除: "${deletedText}"`
    );

    executeCommand(command);
  }, [content, setSelectionRange, executeCommand]);

  // 替换文本命令
  const replaceText = useCallback((start, end, newText) => {
    const actualStart = Math.min(start, end);
    const actualEnd = Math.max(start, end);
    const oldText = content.substring(actualStart, actualEnd);

    const command = new Command(
      // execute
      () => {
        const textarea = textareaRef.current;
        const beforeText = textarea.value.substring(0, actualStart);
        const afterText = textarea.value.substring(actualEnd);
        const newContent = beforeText + newText + afterText;

        textarea.value = newContent;
        setContent(newContent);
        setSelectionRange(actualStart + newText.length, actualStart + newText.length);
      },
      // undo
      () => {
        const textarea = textareaRef.current;
        const beforeText = textarea.value.substring(0, actualStart);
        const afterText = textarea.value.substring(actualStart + newText.length);
        const oldContent = beforeText + oldText + afterText;

        textarea.value = oldContent;
        setContent(oldContent);
        setSelectionRange(actualEnd, actualEnd);
      },
      `替换: "${oldText}" -> "${newText}"`
    );

    executeCommand(command);
  }, [content, setSelectionRange, executeCommand]);

  // 格式化文本
  const formatText = useCallback((formatter) => {
    const currentSelection = getSelection();
    const selectedText = content.substring(currentSelection.start, currentSelection.end);
    const formattedText = formatter(selectedText);

    if (selectedText) {
      replaceText(currentSelection.start, currentSelection.end, formattedText);
    }
  }, [content, getSelection, replaceText]);

  // 键盘快捷键
  const handleKeyDown = useCallback((e) => {
    if (e.ctrlKey || e.metaKey) {
      switch (e.key) {
        case 'z':
          e.preventDefault();
          if (e.shiftKey) {
            redo();
          } else {
            undo();
          }
          break;
        case 'y':
          e.preventDefault();
          redo();
          break;
        case 'b':
          e.preventDefault();
          formatText(text => `**${text}**`);
          break;
        case 'i':
          e.preventDefault();
          formatText(text => `*${text}*`);
          break;
      }
    }
  }, [undo, redo, formatText]);

  // 同步内容状态
  const handleContentChange = useCallback((e) => {
    const newValue = e.target.value;
    const oldValue = content;

    if (newValue.length > oldValue.length) {
      // 插入操作
      const insertedText = newValue.substring(oldValue.length);
      const cursorPos = e.target.selectionStart - insertedText.length;
      insertText(insertedText);
    } else if (newValue.length < oldValue.length) {
      // 删除操作
      const cursorPos = e.target.selectionStart;
      deleteText(cursorPos, cursorPos + (oldValue.length - newValue.length));
    }
  }, [content, insertText, deleteText]);

  // 更新选区
  const handleSelectionChange = useCallback(() => {
    setSelection(getSelection());
  }, [getSelection]);

  // 获取历史记录
  const getHistory = useCallback(() => {
    return historyManagerRef.current.getHistory();
  }, []);

  // 跳转到历史记录
  const jumpToHistory = useCallback((index) => {
    const manager = historyManagerRef.current;
    const textarea = textareaRef.current;

    // 重新执行到指定位置的所有命令
    let tempContent = '';
    manager.history.slice(0, index + 1).forEach(cmd => {
      // 这里需要重新执行命令的逻辑
      // 简化实现，实际应用中需要更复杂的处理
    });

    textarea.value = tempContent;
    setContent(tempContent);
    manager.currentIndex = index;
  }, []);

  return (
    <div className="text-editor">
      <div className="editor-toolbar">
        <button
          onClick={undo}
          disabled={!historyManagerRef.current.canUndo()}
          title="撤销 (Ctrl+Z)"
        >
          ↶ 撤销
        </button>
        <button
          onClick={redo}
          disabled={!historyManagerRef.current.canRedo()}
          title="重做 (Ctrl+Y)"
        >
          ↷ 重做
        </button>

        <div className="toolbar-separator" />

        <button
          onClick={() => formatText(text => `**${text}**`)}
          title="粗体 (Ctrl+B)"
        >
          <strong>B</strong>
        </button>
        <button
          onClick={() => formatText(text => `*${text}*`)}
          title="斜体 (Ctrl+I)"
        >
          <em>I</em>
        </button>

        <div className="toolbar-separator" />

        <button onClick={() => insertText('🙂')}>🙂</button>
        <button onClick={() => insertText('❤️')}>❤️</button>
      </div>

      <div className="editor-content">
        <textarea
          ref={textareaRef}
          value={content}
          onChange={handleContentChange}
          onKeyDown={handleKeyDown}
          onSelect={handleSelectionChange}
          onMouseUp={handleSelectionChange}
          placeholder="开始输入..."
          className="editor-textarea"
        />

        <div className="editor-info">
          <span>字符数: {content.length}</span>
          <span>选区: {selection.start}-{selection.end}</span>
        </div>
      </div>

      <div className="editor-history">
        <h4>历史记录</h4>
        <div className="history-list">
          {getHistory().map((item, index) => (
            <div
              key={index}
              className={`history-item ${item.active ? 'active' : ''}`}
              onClick={() => jumpToHistory(index)}
            >
              {item.description}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
```
